    "# Korsmeyer–Peppas Parameter Identification\n",
    "def extract_kp_parameters(data, temperature):\n",
    "    subset = data[data[\"Temperature\"] == temperature]\n",
    "    X = subset[\"ln_t\"].values.reshape(-1, 1)\n",
    "    y = subset[\"ln_F\"].values\n",
    "    \n",
    "    model = LinearRegression()\n",
    "    model.fit(X, y)\n",
    "    \n",
    "    n = model.coef_[0]\n",
    "    ln_k = model.intercept_\n",
    "    k = np.exp(ln_k)\n",
    "    r2 = model.score(X, y)\n",
    "    \n",
    "    return {\n",
    "        \"Temperature\": temperature,\n",
    "        \"n\": round(n, 3),\n",
    "        \"k\": round(k, 5),\n",
    "        \"R2\": round(r2, 3)\n",
    "    }\n",
    "\n",
    "fit_results = [extract_kp_parameters(df_transport, temp) for temp in [700, 1100]]\n",
    "results_df = pd.DataFrame(fit_results)\n",
    "\n",
    "# Export parameters\n",
    "results_df.to_csv(\"../data/processed/korsmeyer_peppas_parameters.csv\", index=False)\n",
    "results_df"
